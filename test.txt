# GET /404 — перший запит, очікується MISS
curl -i http://127.0.0.1:3000/404

# GET /404 — повторний запит, очікується HIT
curl -sS -D - -o /dev/null http://127.0.0.1:3000/404

# PUT /201 — очікується 201
curl -i -X PUT --data-binary @cache/404.jpg http://127.0.0.1:3000/201

# DELETE /201 — очікується 200
curl -i -X DELETE http://127.0.0.1:3000/201

# POSTі /404 — очікується 405
curl -i -X POST http://127.0.0.1:3000/404

# DELETE /201 після видалення — очікується 404
curl -i -X DELETE http://127.0.0.1:3000/201
